cmake_minimum_required(VERSION 3.24)
project(Helix VERSION 0.2.0 LANGUAGES CXX)

option(HELIX_BUILD_TESTS "Build tests" ON)
option(HELIX_BUILD_BENCHMARKS "Build benchmarks" ON)
option(HELIX_ENABLE_AVX2 "Build AVX2 kernel" OFF)
option(HELIX_ENABLE_AVX512 "Build AVX-512 kernel" OFF)
option(HELIX_ENABLE_SANITIZERS "Enable ASan and UBSan" OFF)
option(HELIX_ENABLE_TSAN "Enable ThreadSanitizer" OFF)
option(HELIX_ENABLE_LTO "Enable interprocedural optimization" OFF)
option(HELIX_ENABLE_CLANG_TIDY "Run clang-tidy while compiling" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

find_package(Threads REQUIRED)
if(HELIX_ENABLE_LTO)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT HELIX_IPO_SUPPORTED OUTPUT HELIX_IPO_ERROR)
  if(NOT HELIX_IPO_SUPPORTED)
    message(FATAL_ERROR "LTO requested but unsupported: ${HELIX_IPO_ERROR}")
  endif()
endif()
if(HELIX_ENABLE_CLANG_TIDY)
  find_program(HELIX_CLANG_TIDY_EXE NAMES clang-tidy REQUIRED)
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
add_library(helix_common src/common.cpp)
add_library(helix_numerics src/core.cpp)
add_library(helix_pricing src/pricing_batch.cpp)
add_library(helix_concurrency src/engine.cpp)
add_library(helix_posix src/runtime.cpp)
add_library(helix_core INTERFACE)
add_library(Helix::numerics ALIAS helix_numerics)
add_library(Helix::common ALIAS helix_common)
add_library(Helix::pricing ALIAS helix_pricing)
add_library(Helix::concurrency ALIAS helix_concurrency)
add_library(Helix::posix ALIAS helix_posix)
foreach(target helix_common helix_numerics helix_pricing helix_concurrency helix_posix)
  target_include_directories(${target} PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>)
  target_link_libraries(${target} PUBLIC Threads::Threads)
  if(HELIX_ENABLE_LTO)
    set_property(TARGET ${target} PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
  endif()
  if(HELIX_ENABLE_CLANG_TIDY)
    set_property(TARGET ${target} PROPERTY CXX_CLANG_TIDY
      "${HELIX_CLANG_TIDY_EXE};--warnings-as-errors=*")
  endif()
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  endif()
endforeach()
target_link_libraries(helix_numerics PUBLIC helix_common)
target_link_libraries(helix_pricing PUBLIC helix_numerics helix_common)
target_link_libraries(helix_concurrency PUBLIC helix_pricing helix_numerics helix_common)
target_link_libraries(helix_posix PUBLIC helix_concurrency helix_numerics)
if(UNIX AND NOT APPLE)
  target_link_libraries(helix_posix PUBLIC atomic)
endif()
target_link_libraries(helix_core INTERFACE helix_posix)
if(HELIX_ENABLE_SANITIZERS AND NOT MSVC)
  foreach(target helix_common helix_numerics helix_pricing helix_concurrency helix_posix)
    target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=address,undefined)
  endforeach()
endif()
if(HELIX_ENABLE_TSAN AND NOT MSVC)
  foreach(target helix_common helix_numerics helix_pricing helix_concurrency helix_posix)
    target_compile_options(${target} PUBLIC -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=thread)
  endforeach()
endif()
if(HELIX_ENABLE_AVX2)
  target_sources(helix_numerics PRIVATE src/avx2.cpp)
  target_compile_definitions(helix_numerics PUBLIC HELIX_HAS_AVX2_KERNEL=1)
  if(MSVC)
    set_source_files_properties(src/avx2.cpp PROPERTIES COMPILE_OPTIONS /arch:AVX2)
  else()
    set_source_files_properties(src/avx2.cpp PROPERTIES COMPILE_OPTIONS -mavx2)
  endif()
endif()
if(HELIX_ENABLE_AVX512)
  target_sources(helix_numerics PRIVATE src/avx512.cpp)
  target_compile_definitions(helix_numerics PUBLIC HELIX_HAS_AVX512_KERNEL=1)
  if(MSVC)
    set_source_files_properties(src/avx512.cpp PROPERTIES COMPILE_OPTIONS /arch:AVX512)
  else()
    set_source_files_properties(src/avx512.cpp PROPERTIES COMPILE_OPTIONS "-mavx512f;-mavx512dq")
  endif()
endif()

foreach(app market_data pricer risk helixctl demo)
  add_executable(helix_${app} apps/${app}_main.cpp)
  target_link_libraries(helix_${app} PRIVATE helix_core)
endforeach()
set_target_properties(helix_helixctl PROPERTIES OUTPUT_NAME helixctl)
add_executable(helix_inspect apps/inspect_main.cpp)
target_link_libraries(helix_inspect PRIVATE helix_posix)
add_executable(helix_cleanup apps/cleanup_main.cpp)
target_link_libraries(helix_cleanup PRIVATE helix_posix)
add_executable(helix_supervisor apps/supervisor_main.cpp)
target_link_libraries(helix_supervisor PRIVATE helix_core)

if(HELIX_BUILD_TESTS)
  enable_testing()
  add_executable(helix_tests tests/tests.cpp)
  target_link_libraries(helix_tests PRIVATE helix_core)
  add_test(NAME helix_unit COMMAND helix_tests)
  if(UNIX AND NOT APPLE)
    add_executable(helix_ipc_stress tests/ipc_stress.cpp)
    target_link_libraries(helix_ipc_stress PRIVATE helix_core)
    add_test(NAME helix_ipc_stress COMMAND helix_ipc_stress)
    set_tests_properties(helix_ipc_stress PROPERTIES TIMEOUT 15)
    add_executable(helix_ipc_failure tests/ipc_failure.cpp)
    target_link_libraries(helix_ipc_failure PRIVATE helix_core)
    add_test(NAME helix_ipc_failure COMMAND helix_ipc_failure)
    set_tests_properties(helix_ipc_failure PROPERTIES TIMEOUT 15)
    add_test(NAME helix_integration_demo COMMAND helix_demo --duration 1 --universe 512 --workers 2)
    set_tests_properties(helix_integration_demo PROPERTIES TIMEOUT 15)
    add_test(NAME helix_backpressure_demo COMMAND helix_demo --duration 1
      --universe 1024 --workers 2 --ring-capacity 2 --drop-policy block)
    set_tests_properties(helix_backpressure_demo PROPERTIES TIMEOUT 15)
    add_test(NAME helix_multi_consumer_demo COMMAND helix_demo --duration 1
      --universe 1024 --workers 2 --ring-capacity 8 --drop-policy block
      --risk-consumers 3)
    set_tests_properties(helix_multi_consumer_demo PROPERTIES TIMEOUT 20)
    add_test(NAME helix_supervisor_fault
      COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/supervisor_fault.sh
                   ${CMAKE_CURRENT_BINARY_DIR})
    set_tests_properties(helix_supervisor_fault PROPERTIES TIMEOUT 20)
    add_test(NAME helix_signal_shutdown
      COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/signal_shutdown.sh
                   ${CMAKE_CURRENT_BINARY_DIR})
    set_tests_properties(helix_signal_shutdown PROPERTIES TIMEOUT 15)
    if(HELIX_ENABLE_TSAN)
      set_tests_properties(
        helix_ipc_stress helix_ipc_failure helix_integration_demo
        helix_backpressure_demo helix_multi_consumer_demo
        helix_supervisor_fault helix_signal_shutdown
        PROPERTIES TIMEOUT 60)
    endif()
  endif()
endif()

set_target_properties(helix_numerics PROPERTIES EXPORT_NAME numerics)
set_target_properties(helix_common PROPERTIES EXPORT_NAME common)
set_target_properties(helix_pricing PROPERTIES EXPORT_NAME pricing)
set_target_properties(helix_concurrency PROPERTIES EXPORT_NAME concurrency)
set_target_properties(helix_posix PROPERTIES EXPORT_NAME posix)
set_target_properties(helix_core PROPERTIES EXPORT_NAME core)
install(TARGETS helix_common helix_numerics helix_pricing helix_concurrency helix_posix helix_core
  EXPORT HelixTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(TARGETS helix_market_data helix_pricer helix_risk helix_helixctl
  helix_demo helix_inspect helix_cleanup helix_supervisor
  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES README.md LICENSE DESTINATION ${CMAKE_INSTALL_DOCDIR})
install(FILES config/helix.example.toml DESTINATION ${CMAKE_INSTALL_SYSCONFDIR}/helix)
install(FILES packaging/helix-market-data.service packaging/helix-pricer.service
  packaging/helix-risk.service DESTINATION ${CMAKE_INSTALL_LIBDIR}/systemd/system)
install(PROGRAMS scripts/perf-stat.sh scripts/capture-benchmarks.sh
  scripts/soak-test.sh scripts/native-validate.sh
  DESTINATION ${CMAKE_INSTALL_BINDIR})
write_basic_package_version_file(
  ${CMAKE_CURRENT_BINARY_DIR}/HelixConfigVersion.cmake
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
configure_package_config_file(cmake/HelixConfig.cmake.in
  ${CMAKE_CURRENT_BINARY_DIR}/HelixConfig.cmake
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Helix)
install(EXPORT HelixTargets FILE HelixTargets.cmake NAMESPACE Helix::
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Helix)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/HelixConfig.cmake
  ${CMAKE_CURRENT_BINARY_DIR}/HelixConfigVersion.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Helix)
if(HELIX_BUILD_BENCHMARKS)
  add_executable(helix_bench benchmarks/bench.cpp)
  target_link_libraries(helix_bench PRIVATE helix_core)
  add_executable(helix_ipc_bench benchmarks/ipc_bench.cpp)
  target_link_libraries(helix_ipc_bench PRIVATE helix_core)
  if(UNIX AND NOT APPLE)
    target_link_libraries(helix_ipc_bench PRIVATE rt)
  endif()
  install(TARGETS helix_bench helix_ipc_bench
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()

set(CPACK_PACKAGE_VENDOR "Helix contributors")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY
  "Low-latency Linux derivatives pricing infrastructure")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_CONTACT "Helix contributors")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
set(CPACK_GENERATOR "TGZ;DEB")
set(CPACK_SOURCE_GENERATOR "TGZ")
set(CPACK_SOURCE_IGNORE_FILES
  "/build[^/]*/;/install[^/]*/;/.git/;/.github/;~$;${CPACK_SOURCE_IGNORE_FILES}")
set(CPACK_DEBIAN_PACKAGE_MAINTAINER "Helix contributors")
set(CPACK_DEBIAN_PACKAGE_SECTION "devel")
find_program(HELIX_FILE_UTILITY file)
if(HELIX_FILE_UTILITY)
  set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
else()
  set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS OFF)
  message(WARNING "'file' utility not found; Debian dependency auto-detection disabled")

set(CPACK_ARCHIVE_FILE_NAME "helix-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}")
